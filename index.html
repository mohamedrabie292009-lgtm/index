using UnityEngine; public class VehicleSwitcher : MonoBehaviour { public GameObject carBMW; public GameObject motorBike; private bool isCarActive = true; void Start() { // نبدأ بالعربية شغالة والموتوسيكل مخفي carBMW.SetActive(true); motorBike.SetActive(false); } void Update() { // الضغط على حرف V للتبديل بين العربية والموتوسيكل if (Input.GetKeyDown(KeyCode.V)) { SwitchVehicle(); } } void SwitchVehicle() { isCarActive = !isCarActive; if (isCarActive) { // انقل العربية لنفس مكان الموتوسيكل وشغلها carBMW.transform.position = motorBike.transform.position; carBMW.transform.rotation = motorBike.transform.rotation; carBMW.SetActive(true); motorBike.SetActive(false); } else { // انقل الموتوسيكل لنفس مكان العربية وشغله motorBike.transform.position = carBMW.transform.position; motorBike.transform.rotation = carBMW.transform.rotation; motorBike.SetActive(true); carBMW.SetActive(false); } } }
